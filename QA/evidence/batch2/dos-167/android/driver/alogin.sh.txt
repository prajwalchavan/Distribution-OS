#!/bin/bash
# Copy of QA/tools/android-login.sh for the DOS-167 Android proof: evidence goes to QA/evidence/batch2/dos-167/android
# (the original writes over tracked phase0 files). Modes:
#   fresh  <user> <tag>  : install -r the debug APK, pm clear, start, sign in (android-login.sh behaviour)
#   signin <user> <tag>  : NO install, NO clear: fill the sign-in form already on screen, tap Sign in,
#                          then 6 screenshots 150 ms apart and ui texts dumps (step 3/4 leak window)
set -u
MODE=$1; USER_=$2; TAG=$3; PORT=5175; PASS='Dos@1234'; APP=sales; TITLE=Sales
ROOT="/Users/prajwalchavan/Desktop/Distribution OS"; T="$ROOT/QA/tools"
OUT="$ROOT/QA/evidence/batch2/dos-167/android"; mkdir -p "$OUT"
export ANDROID_HOME=$HOME/Library/Android/sdk; export PATH="$ANDROID_HOME/platform-tools:$PATH"
PKG="in.distributionos.$APP"
ui() { python3 "$T/ui.py" "$@"; }
for p in 3000 3001 3002 3003 3004 3005 3006 3007 $PORT; do adb reverse tcp:$p tcp:$p >/dev/null; done
if ! curl -s -H "Host: 10.0.2.2:8081" http://127.0.0.1:8081/status | grep -q running; then
  pkill -f "node .*proxy8081.mjs" 2>/dev/null; sleep 1
  nohup node "$T/proxy8081.mjs" $PORT </dev/null >/dev/null 2>&1 & disown
  for i in 1 2 3 4 5; do curl -s -o /dev/null -H "Host: 10.0.2.2:8081" http://127.0.0.1:8081/status && break; sleep 1; done
fi
curl -s -H "Host: 10.0.2.2:8081" http://127.0.0.1:8081/status | grep -q running || { echo "PROXY_DOWN $TAG"; exit 4; }
if [ "$MODE" = fresh ]; then
  adb install -r "$ROOT/frontend/$APP-app/android/app/build/outputs/apk/debug/app-debug.apk" >/dev/null 2>&1 || { echo "INSTALL FAILED $TAG"; exit 1; }
  adb shell am force-stop $PKG; adb shell pm clear $PKG >/dev/null; adb shell am start -n $PKG/.MainActivity >/dev/null 2>&1
fi
for i in $(seq 1 100); do ui has-edit && break; sleep 3; done
if ! ui has-edit; then adb exec-out screencap -p > "$OUT/$TAG-noform.png"; echo "NO_SIGN_IN_FORM $TAG: $(ui texts)"; exit 2; fi
if ! ui texts | grep -q "Distribution OS - $TITLE"; then adb exec-out screencap -p > "$OUT/$TAG-wrongbundle.png"; echo "BUNDLE_MISMATCH $TAG: $(ui texts)"; exit 5; fi
adb exec-out screencap -p > "$OUT/$TAG-signin-form.png"
clear_field() { adb shell input keyevent KEYCODE_MOVE_END; adb shell input keyevent --longpress $(printf 'KEYCODE_DEL %.0s' $(seq 1 40)); }
adb shell input tap $(ui edit 1); sleep 1; clear_field; adb shell input text "$USER_"; sleep 1
adb shell input tap $(ui edit 2); sleep 1; clear_field; adb shell input text "$PASS"; sleep 1
adb shell input keyevent 111; sleep 1
BTN=$(ui text "Sign in")
adb shell input tap $BTN
TAP_AT=$(python3 -c 'import time;print(time.time())')
if [ "$MODE" = signin ]; then
  for i in 1 2 3 4 5 6; do adb exec-out screencap -p > "$OUT/$TAG-$i.png"; echo "$i $(python3 -c 'import time;print(round(time.time()-'"$TAP_AT"',3))')" >> "$OUT/$TAG-shot-times.txt"; sleep 0.15; done
  for i in 1 2 3; do echo "dump$i +$(python3 -c 'import time;print(round(time.time()-'"$TAP_AT"',3))')s: $(ui texts)" >> "$OUT/$TAG-texts-early.txt"; done
fi
signin() { ui texts | grep -q "Use the username\|Invalid username"; }
for i in $(seq 1 25); do sleep 2; signin || break; done
sleep 5; adb exec-out screencap -p > "$OUT/$TAG-home.png"
if signin; then echo "STILL_ON_SIGN_IN $TAG: $(ui texts)"; exit 3; else echo "SIGNED_IN $TAG: $(ui texts)"; fi
