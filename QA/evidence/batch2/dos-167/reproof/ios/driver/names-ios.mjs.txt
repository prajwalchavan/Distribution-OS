const PREFIXES = new Map([['s', 'dos-sales'], ['d', 'dos-delivery'], ['w', 'dos-warehouse'], ['h', 'dos-harness']])
const STORE_NAME = /^([sdwh])([0-9a-z]{25})([0-9a-z]{25})$/
function uuid(digits) {
  let value = 0n
  for (const digit of digits) value = value * 36n + BigInt(Number.parseInt(digit, 36))
  const hex = value.toString(16).padStart(32, '0')
  if (hex.length > 32) return null
  return `${hex.slice(0, 8)}-${hex.slice(8, 12)}-${hex.slice(12, 16)}-${hex.slice(16, 20)}-${hex.slice(20)}`
}
function encode(userId, tenantId, p) {
  const b36 = (u) => BigInt('0x' + u.replace(/-/g, '')).toString(36).padStart(25, '0')
  return p + b36(userId) + b36(tenantId)
}
function parseStoreName(name) {
  const m = STORE_NAME.exec(name)
  if (m === null) return null
  const prefix = PREFIXES.get(m[1]); const userId = uuid(m[2]); const tenantId = uuid(m[3])
  if (prefix === undefined || userId === null || tenantId === null) return null
  return { prefix, userId, tenantId }
}

const names = readFileSync(0, 'utf8').split('\n').map((l) => l.trim()).filter(Boolean)
const set = new Set(names)
const out = { stores: [], strays: [], legacy: [], interim: [], other: [] }
for (const n of names) {
  const base = n.replace(/-(wal|shm|journal)$/, '')
  if (base !== n) { if (!set.has(base)) out.strays.push(n); continue }
  if (/^dos-[a-z]+\.db$/.test(n)) { out.legacy.push(n); continue }
  if (/^dos-[a-z]+__u-/.test(n)) { out.interim.push(n); continue }
  const parsed = parseStoreName(n)
  if (parsed === null) { out.other.push(n); continue }
  const who = Object.keys(IDS).find((w) => IDS[w].userId === parsed.userId && IDS[w].tenantId === parsed.tenantId)
  out.stores.push({ name: n, length: n.length, pattern: /^[sdw][0-9a-z]{50}$/.test(n), ...parsed, who: who ? `${who} @ ${IDS[who].d}` : 'UNKNOWN IDENTITY', reencodes: who ? encode(IDS[who].userId, IDS[who].tenantId, n[0]) === n : false })
}
